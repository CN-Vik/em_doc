# 完整 C 语言 环形队列使用案例（单文件）
文件名：`ring_queue.c`
```c
#include <stdio.h>

// 小猫数据结构（和你之前的小猫信息对应）
typedef struct {
    int id;         // 小猫ID
    float weight;   // 小猫体重
} Cat;

// ===================== 环形队列定义 =====================
#define MAX_SIZE 8  // 队列最大容量（可自己改）

typedef struct {
    Cat data[MAX_SIZE];  // 用数组存小猫数据
    int front;           // 队头：取数据
    int rear;            // 队尾：存数据
} RingQueue;

// ===================== 队列操作 API =====================
// 1. 初始化队列
void queueInit(RingQueue *q) {
    q->front = q->rear = 0;
}

// 2. 判断队列是否为空
int queueIsEmpty(RingQueue *q) {
    return q->front == q->rear;
}

// 3. 判断队列是否已满
int queueIsFull(RingQueue *q) {
    return (q->rear + 1) % MAX_SIZE == q->front;
}

// 4. 入队（添加小猫）
int enQueue(RingQueue *q, Cat cat) {
    if (queueIsFull(q)) {
        printf("队列满了，无法添加！\n");
        return 0;
    }
    q->data[q->rear] = cat;
    q->rear = (q->rear + 1) % MAX_SIZE;
    return 1;
}

// 5. 出队（取出小猫）
int deQueue(RingQueue *q, Cat *outCat) {
    if (queueIsEmpty(q)) {
        printf("队列空的，没有数据！\n");
        return 0;
    }
    *outCat = q->data[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    return 1;
}

// 6. 查看当前队列里有多少只小猫
int queueSize(RingQueue *q) {
    return (q->rear - q->front + MAX_SIZE) % MAX_SIZE;
}

// 7. 遍历打印队列里所有小猫
void showQueue(RingQueue *q) {
    if (queueIsEmpty(q)) {
        printf("队列为空\n");
        return;
    }

    printf("=== 当前队列共有 %d 只小猫 ===\n", queueSize(q));
    int i = q->front;
    while (i != q->rear) {
        printf("小猫ID：%d，体重：%.2f kg\n", q->data[i].id, q->data[i].weight);
        i = (i + 1) % MAX_SIZE;
    }
}

// ===================== 主函数：使用演示 =====================
int main() {
    RingQueue q;
    queueInit(&q);  // 初始化队列

    Cat cat1 = {1001, 2.5f};
    Cat cat2 = {1002, 3.2f};
    Cat cat3 = {1003, 1.9f};
    Cat getCat;  // 用来接收出队的小猫

    printf("===== 开始入队 =====\n");
    enQueue(&q, cat1);
    enQueue(&q, cat2);
    enQueue(&q, cat3);
    showQueue(&q);

    printf("\n===== 出队 1 只 =====\n");
    deQueue(&q, &getCat);
    printf("出队小猫：ID=%d，体重=%.2f\n", getCat.id, getCat.weight);
    showQueue(&q);

    printf("\n===== 再入队 2 只 =====\n");
    Cat cat4 = {1004, 4.1f};
    Cat cat5 = {1005, 2.3f};
    enQueue(&q, cat4);
    enQueue(&q, cat5);
    showQueue(&q);

    return 0;
}
```
