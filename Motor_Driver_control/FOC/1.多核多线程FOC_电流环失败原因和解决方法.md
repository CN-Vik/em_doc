# ⚡ 六、FOC控制函数（完整迁移版）

```c id="s3"
void FOC_Control(void)
{
    // =========================
    // 1. 读取电流（DMA同步）
    // =========================
    float ia = adc.a;
    float ib = adc.b;

    // =========================
    // 2. 电角度（实时）
    // =========================
    float theta = get_angle();
    float s = sinf(theta);
    float c = cosf(theta);

    // =========================
    // 3. Clarke
    // =========================
    float alpha = ia;
    float beta  = (ia + 2*ib) * 0.57735f;

    // =========================
    // 4. Park
    // =========================
    float id = alpha * c + beta * s;
    float iq = beta * c - alpha * s;

    // =========================
    // 5. PI（只控iq）
    // =========================
    float err = iq_ref - iq;

    static float i_term = 0;
    i_term += Ki * err;
    i_term = CLAMP(i_term);

    float vq = Kp * err + i_term;

    // =========================
    // 6. 逆变换
    // =========================
    float alpha_v = -vq * s;
    float beta_v  =  vq * c;

    // =========================
    // 7. SVPWM
    // =========================
    SVPWM(alpha_v, beta_v);
}
```
